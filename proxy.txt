# Настройки локального прокси VPN для плеера.
#
# Нужны, только если плеер не подключается в режиме «proxy» (когда VPN
# работает не через TUN). Значения смотрите в настройках прокси вашего
# VPN-клиента.
#
# Порт и тип можно не указывать — плеер находит их сам. Но если
# автопоиск не справился, задайте вручную:
#
# port = 10808
# type = socks5
#
# Логин и пароль нужны, если клиент требует авторизацию. Чаще всего не
# хватает именно их: без них прокси отвечает «407» или требует пароль,
# и подключения не происходит.
#
# user = ваш_логин
# pass = ваш_пароль
